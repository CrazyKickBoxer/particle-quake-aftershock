cmake_minimum_required(VERSION 3.24)
project(Aftershock LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(PXROOT "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/physx")
file(GLOB BLAST_SOURCES "${PXROOT}/blast/source/sdk/lowlevel/*.cpp" "${PXROOT}/blast/source/sdk/common/*.cpp" "${PXROOT}/blast/source/sdk/globals/*.cpp")
add_library(aftershock_physics STATIC src/aftershock_physics.cpp ${BLAST_SOURCES})
target_include_directories(aftershock_physics PUBLIC src PRIVATE
 "${PXROOT}/physx/include" "${PXROOT}/blast/include/lowlevel" "${PXROOT}/blast/include/shared" "${PXROOT}/blast/include/shared/NvFoundation" "${PXROOT}/blast/include/globals"
 "${PXROOT}/blast/source/sdk/lowlevel" "${PXROOT}/blast/source/sdk/common" "${PXROOT}/blast/source/shared" "${PXROOT}/blast/source/shared/NsFoundation/include")
target_compile_definitions(aftershock_physics PRIVATE PX_PHYSX_STATIC_LIB NVBLAST_STATIC_LIB NDEBUG _CRT_SECURE_NO_WARNINGS)
target_compile_options(aftershock_physics PRIVATE /W3 /utf-8)
file(GLOB_RECURSE PX_LIBS "${CMAKE_CURRENT_SOURCE_DIR}/build/physx/lib/*.lib")
list(FILTER PX_LIBS INCLUDE REGEX "/release/")
target_link_libraries(aftershock_physics PUBLIC ${PX_LIBS})
add_executable(aftershock_physics_tests tests/physics_tests.cpp)
target_link_libraries(aftershock_physics_tests PRIVATE aftershock_physics)
enable_testing()
add_test(NAME structural_physics COMMAND aftershock_physics_tests)
add_executable(aftershock_pressure_tests tests/pressure.cpp)
target_link_libraries(aftershock_pressure_tests PRIVATE aftershock_physics)
add_test(NAME physics_pressure COMMAND aftershock_pressure_tests)
add_executable(aftershock_contract_tests tests/contracts.cpp)
target_include_directories(aftershock_contract_tests PRIVATE src)
add_test(NAME gpu_contracts COMMAND aftershock_contract_tests "${CMAKE_CURRENT_SOURCE_DIR}/build")
add_executable(aftershock_gib_tests tests/gib_motion.cpp)
target_include_directories(aftershock_gib_tests PRIVATE src)
add_test(NAME gib_motion COMMAND aftershock_gib_tests)
add_executable(holo_physics_contract_tests tests/holo_physics_contract.cpp)
target_include_directories(holo_physics_contract_tests PRIVATE src)
add_test(NAME holo_physics_contract COMMAND holo_physics_contract_tests)
add_executable(aftershock_arena_generator tests/arena_generator.cpp)
target_compile_options(aftershock_arena_generator PRIVATE /utf-8)
add_executable(vkquake_launcher WIN32 src/launcher.cpp)
target_compile_options(vkquake_launcher PRIVATE /utf-8)
target_link_libraries(vkquake_launcher PRIVATE user32 gdi32 shell32 ole32 advapi32)
